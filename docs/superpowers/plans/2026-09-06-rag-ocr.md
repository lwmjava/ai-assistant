# RAG OCR 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为扫描版 PDF 增加 OCR 识别能力，在现有 RAG 导入平台中支持本地 Tesseract 默认实现，并为云 OCR provider 预留统一扩展接口。

**Architecture:** 在 `app/rag/ocr/` 下新增 OCR 抽象、工厂与 Tesseract provider，`app/rag/document_parsers/pdf.py` 仍先尝试文本层提取，仅在无文本时调用 OCR。导入任务平台不新增状态机分支，继续通过现有 `ImportJob` 的失败治理、重试与重解析承接 OCR 结果。

**Tech Stack:** Python 3.11, FastAPI, pytest, pypdf, subprocess, Tesseract OCR

from app.core.config import settings
from app.rag.ocr.base import OcrProvider, OcrProviderError
from app.rag.ocr.tesseract import TesseractOcrProvider

def get_ocr_provider() -> OcrProvider:
    if not settings.RAG_OCR_ENABLED:
        raise OcrProviderError("该 PDF 需要 OCR 才能提取文本，但当前系统未启用 OCR")
    if settings.RAG_OCR_PROVIDER == "tesseract":
        return TesseractOcrProvider(
            languages=settings.RAG_OCR_LANGUAGES,
            timeout_seconds=settings.RAG_OCR_TIMEOUT_SECONDS,
        )
    if settings.RAG_OCR_PROVIDER == "cloud":
        raise OcrProviderError("当前 OCR provider 未实现，请先配置可用的本地或云 OCR 服务")
    raise OcrProviderError(f"不支持的 OCR provider: {settings.RAG_OCR_PROVIDER}")

# app/rag/ocr/tesseract.py
import shutil
import subprocess

from app.rag.ocr.base import OcrProvider, OcrProviderError, OcrResult

def _render_pdf_pages(file_bytes: bytes) -> list[bytes]:
    return [file_bytes]

class TesseractOcrProvider(OcrProvider):
    def __init__(self, *, languages: str, timeout_seconds: float) -> None:
        self.languages = languages
        self.timeout_seconds = timeout_seconds

    def extract_pdf_text(self, file_bytes: bytes) -> OcrResult:
        binary = shutil.which("tesseract")
        if not binary:
            raise OcrProviderError("OCR 依赖缺失：未检测到 tesseract 或 chi_sim/eng 语言包")
        texts: list[str] = []
        try:
            for image_bytes in _render_pdf_pages(file_bytes):
                proc = subprocess.run(
                    [binary, "stdin", "stdout", "-l", self.languages],
                    input=image_bytes,
                    capture_output=True,
                    text=True,
                    timeout=self.timeout_seconds,
                    check=False,
                )
                if proc.returncode != 0:
                    raise OcrProviderError("OCR 执行失败，请检查服务器 OCR 环境或稍后重试")
                texts.append(proc.stdout.strip())
        except subprocess.TimeoutExpired as exc:
            raise OcrProviderError("OCR 执行失败，请检查服务器 OCR 环境或稍后重试") from exc
        text = "\n".join(part for part in texts if part).strip()
        if not text:
            raise OcrProviderError("OCR 已执行，但未提取到可用文本")
        return OcrResult(text=text, provider="tesseract", used_ocr=True)
```

from app.rag.document_parsers.base import (
    DocumentOcrRequiredError,
    DocumentParseError,
    DocumentParser,
    ParsedDocument,
)
from app.rag.ocr.base import OcrProviderError
from app.rag.ocr.factory import get_ocr_provider

class PdfDocumentParser(DocumentParser):
    def __init__(self, ocr_provider=None) -> None:
        self._ocr_provider = ocr_provider

    def extract(self, file_bytes: bytes, filename: str, content_type: str | None) -> ParsedDocument:
        reader = PdfReader(io.BytesIO(file_bytes))
        if reader.is_encrypted:
            raise DocumentParseError("文件解析失败，请确认文件未损坏或未加密")
        texts = [(page.extract_text() or "").strip() for page in reader.pages]
        text = "\n".join(part for part in texts if part)
        if text:
            return ParsedDocument(
                text=text,
                title=title_from_filename(filename),
                source=filename,
                extension="pdf",
                content_type=content_type,
                metadata={"parser_name": "pdf", "used_ocr": False},
            )
        try:
            provider = self._ocr_provider or get_ocr_provider()
            result = provider.extract_pdf_text(file_bytes)
        except OcrProviderError as exc:
            message = str(exc)
            if "未启用 OCR" in message:
                raise DocumentOcrRequiredError(message) from exc
            raise DocumentParseError(message) from exc
        return ParsedDocument(
            text=result.text,
            title=title_from_filename(filename),
            source=filename,
            extension="pdf",
            content_type=content_type,
            metadata={"parser_name": "pdf_ocr", "used_ocr": True, "ocr_provider": result.provider},
        )
```
